1️⃣ PING - Host Discovery

📌 هدف: بررسی زنده بودن هاست و دریافت اطلاعات اولیه

Base دستورات پایه

bash 
# Ping ساده (4 بار)
ping example.com

# Ping با تعداد مشخص
ping -c 10 example.com

# Ping با فاصله زمانی مشخص (ثانیه)
ping -i 0.5 example.com

# Ping با اندازه بسته مشخص
ping -s 1024 example.com

# Ping به آی‌پی
ping 192.168.1.10

# Ping بی‌نهایت (Ctrl+C برای متوقف کردن)
ping -t example.com

Combined دستورات ترکیبی

bash 
# Ping با TTL مشخص
ping -t 64 example.com

# Ping با Don't Fragment
ping -M do example.com

# Ping با Flood (نیاز به root)
sudo ping -f example.com

# Ping با رکورد مسیر
ping -R example.com

# Ping با Timestamp
ping -T example.com

CTF سناریوهای CTF

bash 
# 1. دریافت IP هدف
ping -c 1 example.com | grep -oP '\([0-9.]+\)' | tr -d '()'

# 2. بررسی TTL (تشخیص OS)
ping -c 1 example.com | grep -o 'ttl=[0-9]*'

# 3. بررسی Packet Loss
ping -c 10 example.com | grep -o '[0-9]*% packet loss'

# 4. دریافت Average Time
ping -c 5 example.com | grep -o 'avg/[0-9.]*' | cut -d'/' -f2

# 5. Ping به همه دستگاه‌های شبکه
for i in {1..254}; do ping -c 1 -W 1 192.168.1.$i | grep "64 bytes" & done

# 6. Ping با آرام‌تر (کمتر تهاجمی)
ping -c 4 -i 2 example.com

# 7. تشخیص OS از TTL
ttl=$(ping -c 1 example.com | grep -o 'ttl=[0-9]*' | cut -d'=' -f2)
if [ $ttl -le 64 ]; then echo "Linux/Unix"; elif [ $ttl -le 128 ]; then echo "Windows"; else echo "Unknown"; fi

Tip نکات مهم CTF

bash 
# 1. Ping Sweep (پیدا کردن همه هاست‌های زنده)
nmap -sn 192.168.1.0/24

# 2. Ping با پروتکل TCP (اگر ICMP blocked باشد)
sudo hping3 -S -p 80 example.com

# 3. Ping با پروتکل UDP
sudo hping3 -2 -p 53 example.com

# 4. Ping با استفاده از ARP (در LAN)
arping -c 3 192.168.1.10

# 5. Ping با استفاده از HTTP (اگر ICMP blocked باشد)
curl -s -o /dev/null -w "%{http_code}\n" example.com

# 6. Ping همه هاست‌ها با fping
fping -a -g 192.168.1.0/24

# 7. Ping با IP Spoofing
sudo hping3 -S -p 80 example.com -a 192.168.1.100

2️⃣ HOST - DNS Lookup

📌 هدف: جستجوی ساده DNS

Base دستورات پایه و ترکیبی

bash 
# A Record (IPv4)
host example.com

# AAAA Record (IPv6)
host -t AAAA example.com

# MX Record (Mail)
host -t MX example.com

# NS Record (Name Server)
host -t NS example.com

# TXT Record
host -t TXT example.com

# همه رکوردها
host -t ANY example.com

# استفاده از سرور DNS خاص
host example.com 8.8.8.8

# Reverse DNS
host 8.8.8.8

# با جزئیات بیشتر
host -v example.com

CTF سناریوهای CTF

bash 
# 1. دریافت IP
host example.com | grep "has address" | awk '{print $4}'

# 2. دریافت همه IPها
host example.com | grep "has address" | awk '{print $4}' | sort -u

# 3. دریافت MX
host -t MX example.com | awk '{print $NF}'

# 4. دریافت NS
host -t NS example.com | awk '{print $NF}'

# 5. دریافت SPF از TXT
host -t TXT example.com | grep -i "v=spf1"

# 6. پیدا کردن CNAME
host -t CNAME www.example.com

# 7. Reverse DNS روی IP
host 192.168.1.10

# 8. دریافت همه MX
host -t MX example.com | grep "mail" | awk '{print $NF}'

# 9. دریافت TXT کامل
host -t TXT example.com | sed 's/.*"\(.*\)"/\1/'

# 10. ترکیب با dig برای دریافت جزئیات
host -t MX example.com | while read mx; do
    domain=$(echo $mx | awk '{print $NF}')
    host $domain
done

3️⃣ NSLOOKUP - DNS Lookup تعاملی

📌 هدف: جستجوی DNS با قابلیت تعاملی

Base حالت غیرتعاملی و تعاملی

bash 
# ==================== حالت غیرتعاملی ====================

# جستجوی ساده
nslookup example.com

# جستجوی نوع خاص
nslookup -type=MX example.com
nslookup -type=NS example.com
nslookup -type=TXT example.com
nslookup -type=CNAME example.com
nslookup -type=SOA example.com
nslookup -type=ANY example.com

# استفاده از سرور DNS خاص
nslookup example.com 8.8.8.8

# Reverse DNS
nslookup 8.8.8.8

# ==================== حالت تعاملی ====================

# ورود به حالت تعاملی
nslookup

# دستورات داخل محیط تعاملی
> set type=MX
> example.com
> set type=NS
> example.com
> set type=TXT
> example.com
> set type=ANY
> example.com
> server 8.8.8.8
> example.com
> set debug
> example.com
> exit

CTF سناریوهای CTF

bash 
# 1. دریافت IP
nslookup example.com | grep "Address" | tail -1 | awk '{print $2}'

# 2. دریافت MX
nslookup -type=MX example.com | grep "mail exchanger" | awk '{print $6}'

# 3. دریافت NS
nslookup -type=NS example.com | grep "nameserver" | awk '{print $NF}'

# 4. دریافت TXT
nslookup -type=TXT example.com | grep "text" | sed 's/.*"\(.*\)"/\1/'

# 5. دریافت CNAME
nslookup -type=CNAME www.example.com | grep "canonical name" | awk '{print $NF}'

# 6. تشخیص SPF
nslookup -type=TXT example.com | grep -i "v=spf1"

# 7. دریافت SOA
nslookup -type=SOA example.com | grep "origin" | awk '{print $3}'

# 8. دریافت همه MX با اولویت
nslookup -type=MX example.com | grep "mail exchanger" | while read line; do
    priority=$(echo $line | awk '{print $5}')
    server=$(echo $line | awk '{print $6}')
    echo "$priority $server"
done

# 9. استفاده از سرور DNS مختلف
nslookup example.com 1.1.1.1

# 10. دریافت اطلاعات کامل
nslookup -type=ANY example.com | grep -v "Non-authoritative"

4️⃣ DIG - Advanced DNS

📌 هدف: جستجوی پیشرفته و کامل DNS

Base دستورات پایه و ترکیبی

bash 
# جستجوی کامل
dig example.com

# فقط پاسخ کوتاه
dig example.com +short

# A Record
dig example.com A +short

# AAAA Record
dig example.com AAAA +short

# MX Record
dig example.com MX +short

# NS Record
dig example.com NS +short

# TXT Record
dig example.com TXT +short

# همه رکوردها
dig example.com ANY

# با نمایش TTL
dig example.com +ttlid

# با جزئیات کامل
dig example.com +trace

# استفاده از سرور DNS خاص
dig @8.8.8.8 example.com

# فقط پاسخ
dig example.com +noall +answer

# فقط AUTHORITY
dig example.com +noall +authority

# فقط ADDITIONAL
dig example.com +noall +additional

CTF سناریوهای CTF

bash 
# 1. دریافت IP کوتاه
dig example.com +short

# 2. دریافت همه IPها
dig example.com A +short | sort -u

# 3. دریافت MX با اولویت
dig example.com MX +short | awk '{print $1, $2}'

# 4. دریافت NS
dig example.com NS +short

# 5. دریافت SPF از TXT
dig example.com TXT +short | grep -i "v=spf1"

# 6. دریافت DKIM
dig default._domainkey.example.com TXT +short

# 7. دریافت DMARC
dig _dmarc.example.com TXT +short

# 8. دریافت CNAME
dig www.example.com CNAME +short

# 9. Reverse DNS
dig -x 8.8.8.8 +short

# 10. Zone Transfer (اگر باز باشد)
dig example.com AXFR @ns1.example.com

# 11. دریافت TTL
dig example.com A +noall +answer | awk '{print $2}'

# 12. دریافت همه MX با dig
dig example.com MX +short | while read line; do
    priority=$(echo $line | awk '{print $1}')
    server=$(echo $line | awk '{print $2}')
    ip=$(dig $server A +short | head -1)
    echo "$priority $server -> $ip"
done

# 13. دریافت همه اطلاعات
dig example.com +noall +answer +authority +additional

# 14. دریافت SOA
dig example.com SOA +short | awk '{print $1}'

# 15. دریافت SRV
dig _sip._tcp.example.com SRV +short

Tip نکات مهم CTF

bash 
# 1. Bruteforce زیردامنه
for sub in www mail ftp ssh admin test dev api; do
    dig $sub.example.com A +short | while read ip; do
        if [ ! -z "$ip" ]; then
            echo "[+] $sub.example.com -> $ip"
        fi
    done
done

# 2. دریافت همه رکوردها با فیلتر
dig example.com ANY | grep -E "A|MX|NS|TXT|CNAME|SOA"

# 3. مقایسه DNS مختلف
for dns in 8.8.8.8 1.1.1.1 9.9.9.9; do
    echo "=== $dns ==="
    dig @$dns example.com A +short
done

# 4. دریافت DNSSEC
dig example.com A +dnssec

# 5. دریافت با حداقل پاسخ
dig example.com +nocomments +noquestion +noauthority +noadditional +nostats

# 6. دریافت CDN تشخیص
dig example.com A +short | while read ip; do
    whois $ip | grep -i "cloudflare\|akamai\|fastly\|cloudfront"
done

# 7. دریافت همه MX با IP
dig example.com MX +short | awk '{print $2}' | while read mx; do
    echo "$mx -> $(dig $mx A +short | head -1)"
done

# 8. دریافت اطلاعات کامل در یک فایل
dig example.com +trace > dns_trace.txt

5️⃣ TRACEROUTE - مسیریابی شبکه

📌 هدف: مشاهده مسیر رسیدن به Target

Base دستورات پایه و ترکیبی

bash 
# Traceroute ساده
traceroute example.com

# Traceroute با ICMP
traceroute -I example.com

# Traceroute با TCP
traceroute -T example.com

# با نمایش IP فقط
traceroute -n example.com

# با تنظیم TTL اولیه
traceroute -f 5 example.com

# با تنظیم حداکثر TTL
traceroute -m 20 example.com

# با زمان انتظار کمتر
traceroute -w 1 example.com

# با UDP
traceroute -U example.com

# با مشخص کردن پورت
traceroute -p 80 example.com

# با نمایش AS شماره
traceroute -A example.com

CTF سناریوهای CTF

bash 
# 1. دریافت مسیر
traceroute example.com | grep -v "\\*"

# 2. دریافت فقط IPها
traceroute -n example.com | grep -v "\\*" | awk '{print $2}'

# 3. دریافت تعداد هاپ‌ها
traceroute -n example.com | wc -l

# 4. دریافت IP نهایی
traceroute -n example.com | tail -1 | awk '{print $2}'

# 5. Traceroute با ICMP
traceroute -I example.com

# 6. دریافت مسیر با TCP
traceroute -T -p 80 example.com

# 7. دریافت مسیر با UDP
traceroute -U -p 53 example.com

# 8. دریافت با 64 بیت
traceroute -q 3 example.com

# 9. دریافت با حداقل زمان
traceroute -w 1 -q 1 example.com

# 10. دریافت AS اطلاعات
traceroute -A example.com | grep -o "AS[0-9]*"

6️⃣ NMAP - مهم‌ترین ابزار Active Recon ⭐⭐⭐⭐⭐

📌 هدف: اسکن پورت و Enumeration

6.1 Host Discovery

bash 
# Ping Sweep (ICMP)
nmap -sn 192.168.1.0/24

# Ping Sweep با TCP SYN
nmap -sn -PS 80 192.168.1.0/24

# Ping Sweep با TCP ACK
nmap -sn -PA 80 192.168.1.0/24

# Ping Sweep با UDP
nmap -sn -PU 53 192.168.1.0/24

# Ping Sweep با ARP (LAN)
nmap -sn -PR 192.168.1.0/24

# استفاده از چند روش
nmap -sn -PS 80 -PA 22 -PU 53 192.168.1.0/24

# بدون Ping (فرض کن زنده است)
nmap -Pn 192.168.1.10

# فقط هاست‌های زنده
nmap -sn 192.168.1.0/24 | grep "Nmap scan" | awk '{print $5}'

# بررسی یک هاست
nmap -sn 192.168.1.10

# 1. پیدا کردن همه هاست‌های زنده
nmap -sn 192.168.1.0/24 | grep -B 2 "Host is up" | grep "Nmap scan" | awk '{print $5}'

# 2. پیدا کردن IP یک دامنه
nmap -sn example.com

# 3. پیدا کردن هاست‌های با پورت خاص
nmap -sn -PS 22 192.168.1.0/24

# 4. پیدا کردن همه هاست‌های با پاسخ
nmap -sn -PR 192.168.1.0/24 | grep "Host is up" -B 2 | grep "Nmap scan"

# 5. پیدا کردن هاست‌های با IP خاص
nmap -sn 192.168.1.1-50

6.2 Port Scanning

bash 
# ==================== تکنیک‌های اسکن ====================

# UDP Scan
nmap -sU 192.168.1.10

nmap -sF 192.168.1.10

# NULL Scan
nmap -sN 192.168.1.10

# XMAS Scan
nmap -sX 192.168.1.10

# ACK Scan (Firewall Detection)
nmap -sA 192.168.1.10

# Window Scan
nmap -sW 192.168.1.10

# Maimon Scan
nmap -sM 192.168.1.10

# ==================== دستورات ترکیبی ====================

# اسکن سریع (پورت‌های Top 100)
nmap -T4 -F 192.168.1.10

# اسکن پورت‌های خاص

# اسکن رنج پورت
nmap -p 1-1000 192.168.1.10

# اسکن همه پورت‌ها
nmap -p- 192.168.1.10

# اسکن با محدودیت زمان

# ==================== سناریوهای CTF ====================

nmap -T4 -F 192.168.1.10

# 2. اسکن همه پورت‌ها

# 6. اسکن با تحمل خطا

# 7. اسکن با تاخیر
nmap -p- --min-rate 1000 192.168.1.10

# 8. اسکن UDP روی پورت‌های رایج
nmap -sU -p 53,161,137,138,139 192.168.1.10

# 9. اسکن با پورت‌های تصادفی
nmap -p 80 --randomize-hosts 192.168.1.10

# 10. اسکن با پورت‌های Top 1000
nmap --top-ports 1000 192.168.1.10

📝 اسکریپت کامل Active Recon Auto

📌 هدف: اجرای خودکار همه ابزارها برای جمع‌آوری سریع اطلاعات
bash 
#!/bin/bash
# active_cheatsheet.sh - اسکریپت کامل Active Recon

TARGET=${1:-192.168.1.10}
OUTPUT_DIR="active_recon_$(date +%Y%m%d_%H%M%S)"
mkdir -p $OUTPUT_DIR

echo "========================================="
echo "  ACTIVE RECON CHEAT SHEET AUTO-RUN"
echo "========================================="

# 1. Ping
echo "[1/12] Running Ping..."
ping -c 4 $TARGET > $OUTPUT_DIR/01_ping.txt
echo "✓ Ping saved"

# 2. Host
echo "[2/12] Running Host..."
host $TARGET > $OUTPUT_DIR/02_host.txt
host -t MX $TARGET >> $OUTPUT_DIR/02_host.txt
host -t NS $TARGET >> $OUTPUT_DIR/02_host.txt
host -t TXT $TARGET >> $OUTPUT_DIR/02_host.txt
echo "✓ Host saved"

# 3. nslookup
echo "[3/12] Running nslookup..."
nslookup $TARGET > $OUTPUT_DIR/03_nslookup.txt
nslookup -type=MX $TARGET >> $OUTPUT_DIR/03_nslookup.txt
nslookup -type=NS $TARGET >> $OUTPUT_DIR/03_nslookup.txt
nslookup -type=TXT $TARGET >> $OUTPUT_DIR/03_nslookup.txt
echo "✓ nslookup saved"

# 4. dig
echo "[4/12] Running dig..."
for type in A AAAA MX NS TXT CNAME SOA; do
    dig $TARGET $type +short > $OUTPUT_DIR/04_dig_$type.txt
done
echo "✓ dig saved"

# 5. traceroute
echo "[5/12] Running traceroute..."
traceroute -n $TARGET > $OUTPUT_DIR/05_traceroute.txt
echo "✓ traceroute saved"

# 6. Nmap Basic
echo "[6/12] Running Nmap Basic..."
nmap -T4 -F $TARGET > $OUTPUT_DIR/06_nmap_basic.txt
echo "✓ Nmap Basic saved"

# 7. Nmap Full Ports
echo "[7/12] Running Nmap Full Ports..."
nmap -p- -T4 $TARGET > $OUTPUT_DIR/07_nmap_full.txt
echo "✓ Nmap Full saved"

# 8. Nmap Services
echo "[8/12] Running Nmap Services..."
nmap -sV -p 22,25,53,80,110,111,135,139,143,443,445,993,995,1723,3306,3389,5900,8080 $TARGET > $OUTPUT_DIR/08_nmap_services.txt
echo "✓ Nmap Services saved"

# 9. Nmap OS
echo "[9/12] Running Nmap OS..."
nmap -O $TARGET > $OUTPUT_DIR/09_nmap_os.txt
echo "✓ Nmap OS saved"

# 10. Nmap Scripts
echo "[10/12] Running Nmap Scripts..."
nmap -sC -p 22,80,443 $TARGET > $OUTPUT_DIR/10_nmap_scripts.txt
echo "✓ Nmap Scripts saved"

# 11. Nmap Aggressive
echo "[11/12] Running Nmap Aggressive..."
nmap -A -T4 $TARGET > $OUTPUT_DIR/11_nmap_aggressive.txt
echo "✓ Nmap Aggressive saved"

# 12. Masscan
echo "[12/12] Running Masscan..."
if command -v masscan &> /dev/null; then
    masscan -p1-65535 $TARGET --rate=1000 > $OUTPUT_DIR/12_masscan.txt
    echo "✓ Masscan saved"
else
    echo "✗ Masscan not installed"
fi

echo ""
echo "========================================="
echo "  ✅ RECON COMPLETE"
echo "========================================="
echo "📁 Output: $OUTPUT_DIR"
ls -la $OUTPUT_DIR
