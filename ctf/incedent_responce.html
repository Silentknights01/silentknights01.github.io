📊 فاز ۱: تحلیل گزارش‌ها (Log Analysis)

۱. تحلیل لاگ‌های لینوکس (auth.log)

bash 
# لاگ‌های SSH
cat /var/log/auth.log | grep sshd
cat /var/log/auth.log | grep "Failed password"
cat /var/log/auth.log | grep "Accepted password"
cat /var/log/auth.log | grep "Invalid user"

# Brute Force Detection
grep "Failed password" /var/log/auth.log | awk '{print $(NF-3)}' | sort | uniq -c | sort -nr
grep "Failed password" /var/log/auth.log | awk -F'from ' '{print $2}' | awk '{print $1}' | sort | uniq -c | sort -nr

# Successful logins بعد از failures
grep "Accepted password" /var/log/auth.log | awk '{print $1,$2,$3,$9,$11}'

# SSH key login
grep "Accepted publickey" /var/log/auth.log

# دستورات sudo اجرا شده
grep "sudo" /var/log/auth.log
grep "COMMAND" /var/log/auth.log

# کاربران جدید و تغییرات پسورد
grep "new user" /var/log/auth.log
grep "useradd" /var/log/auth.log
grep "password changed" /var/log/auth.log

۲. تحلیل لاگ‌های وب‌سرور (Apache/Nginx)

bash 
# Apache access log
cat /var/log/apache2/access.log | grep -E "POST|PUT|DELETE"
cat /var/log/apache2/access.log | awk '{print $1}' | sort | uniq -c | sort -nr  # IP ها

# پیدا کردن Web Shell
cat /var/log/apache2/access.log | grep -E "\.php|\.asp|\.aspx|\.jsp"
cat /var/log/apache2/access.log | grep -E "(cmd|exec|system|shell|eval)"
cat /var/log/apache2/access.log | grep -E "(whoami|id|cat /etc/passwd)"

# پیدا کردن فایل‌های آپلود شده
cat /var/log/apache2/access.log | grep -E "(\.jpg|\.png|\.gif|\.php|\.asp)" | grep "POST"

# Nginx
cat /var/log/nginx/access.log
cat /var/log/nginx/error.log

🕵️ فاز ۲: شناسایی شاخص‌های آلودگی (IOCs)

۱. فرآیندهای عجیب و اتصالات شبکه

bash 
# لیست فرآیندهای مخفی (لینوکس)
ps aux | grep -E "(nc|netcat|telnet|ssh|bash -i|python -c|perl -e|php -r)"
ps auxf
pstree -p

# فرآیندهای با اتصال شبکه
lsof -i
netstat -tulpn | grep LISTEN
ss -tulpn

# اتصالات خارجی
netstat -antp | grep ESTABLISHED
ss -antp | grep ESTABLISHED

# IP های مخرب شناخته شده
netstat -antp | grep -E "(evil|malware|1337|4444|31337)"

# نظارت بر اتصالات
iftop -i eth0
nethogs eth0

۲. وظایف زمان‌بندی شده (Cron/Scheduled Tasks)

bash 
# بررسی cron (لینوکس)
cat /etc/crontab
cat /etc/cron.d/*
crontab -l
crontab -l -u root

# پیدا کردن cron های جدید
find /etc -name "*cron*" -type f -mtime -7 2>/dev/null

# بررسی systemd timers
systemctl list-timers --all
powershell 
# بررسی Scheduled Tasks (ویندوز)
schtasks /query /fo LIST /v
Get-ScheduledTask | Where-Object {$_.State -eq "Running"}

# پیدا کردن tasks جدید
Get-ScheduledTask | Where-Object {$_.Date -gt (Get-Date).AddDays(-7)}

۳. فایل‌های مخرب و Web Shell ها

bash 
# فایل‌های جدید (لینوکس)
find / -type f -mtime -7 2>/dev/null | grep -v "/proc\|/sys"
find / -type f -mmin -60 2>/dev/null

# فایل‌های با مجوز SUID جدید
find / -type f -perm -4000 -mtime -30 2>/dev/null

# Web Shell ها
find /var/www -name "*.php" -mtime -7 2>/dev/null
grep -r "system(" /var/www 2>/dev/null
grep -r "eval(" /var/www 2>/dev/null
grep -r "shell_exec(" /var/www 2>/dev/null

# فایل‌های SSH
ls -la ~/.ssh/
cat ~/.ssh/authorized_keys
powershell 
# فایل‌های اجرایی جدید (ویندوز)
Get-ChildItem -Path C:\ -Recurse -Include *.exe,*.dll,*.ps1,*.bat,*.cmd -File -ErrorAction SilentlyContinue | Where-Object {$_.CreationTime -gt (Get-Date).AddDays(-7)}

# فایل‌های مخفی
Get-ChildItem -Path C:\ -Recurse -Force -File -ErrorAction SilentlyContinue | Where-Object {$_.Attributes -match "Hidden"}

# Startup folders
Get-ChildItem "C:\ProgramData\Microsoft\Windows\Start Menu\Programs\StartUp"
Get-ChildItem "C:\Users\$env:USERNAME\AppData\Roaming\Microsoft\Windows\Start Menu\Programs\Startup"

📅 فاز ۳: بازسازی خط زمانی (Timeline Analysis)

۱. ایجاد خط زمانی لینوکس

bash 
# جمع‌آوری تمام رویدادها در یک فایل
echo "=== Timeline Analysis ===" > timeline.txt

echo "=== System Logs ===" >> timeline.txt
cat /var/log/syslog >> timeline.txt

echo "=== Authentication Logs ===" >> timeline.txt
cat /var/log/auth.log >> timeline.txt

echo "=== Web Access Logs ===" >> timeline.txt
cat /var/log/apache2/access.log >> timeline.txt

echo "=== New Files ===" >> timeline.txt
find / -type f -mtime -30 2>/dev/null >> timeline.txt

# مرتب‌سازی بر اساس زمان
sort -k1 timeline.txt > sorted_timeline.txt

# پیدا کردن نقاط عطف
grep -E "(Accepted password|sudo|useradd|new user|service started)" timeline.txt

# یافتن اولین دسترسی
grep "Accepted password" timeline.txt | head -1
grep "Failed password" timeline.txt | head -5

۲. بازسازی زنجیره حمله

bash 
echo "=== Attack Chain Reconstruction ===" > attack_chain.txt
echo "1. Initial Access:" >> attack_chain.txt
grep "Accepted password" timeline.txt | head -1 >> attack_chain.txt

echo "2. Privilege Escalation:" >> attack_chain.txt
grep "sudo" timeline.txt | head -5 >> attack_chain.txt

echo "3. Persistence:" >> attack_chain.txt
grep -E "cron|service|useradd" timeline.txt | head -5 >> attack_chain.txt

echo "4. Lateral Movement:" >> attack_chain.txt
grep -E "ssh.*from|scp|rsync" timeline.txt | head -5 >> attack_chain.txt

echo "5. Data Exfiltration:" >> attack_chain.txt
grep -E "ftp|scp|curl.*POST" timeline.txt | head -5 >> attack_chain.txt

🦠 فاز ۴: تحلیل بدافزار (Malware Triage)

۱. تحلیل استاتیک پایه

bash 
# اطلاعات پایه و هش
file malware.bin
md5sum malware.bin
sha256sum malware.bin

# Strings
strings malware.bin | head -100
strings malware.bin | grep -E "(http|https|cmd|powershell|bash|python)"
strings -n 10 malware.bin > strings.txt

# بخش‌ها (ELF)
readelf -h malware.bin
readelf -S malware.bin

# با objdump و nm
objdump -d malware.bin | head -100
nm malware.bin
ldd malware.bin

۲. تحلیل با Radare2 و Auditd

bash 
# تحلیل با Radare2
r2 malware.bin
> aaa  # تحلیل پیشرفته
> izz  # جستجوی رشته‌ها
> ii   # بررسی import/export
> pdf @ main  # دیساسمبلی

# با strace (اجرا و مانیتورینگ)
strace -e trace=file,network,process malware.bin
strace -o trace.log malware.bin

# با auditd (لینوکس)
ausearch -ts today
ausearch -i -p 1234

۳. تحلیل PowerShell Scripts

powershell 
# بررسی اسکریپت
Get-Content malware.ps1

# جستجوی دستورات مخرب
Select-String -Path malware.ps1 -Pattern "Invoke-Expression|IEX|DownloadFile|Invoke-WebRequest|Start-Process"

# اجرای با محدودیت
powershell -ExecutionPolicy Bypass -File malware.ps1 -WhatIf

✅ چک‌لیست نهایی CTF IR

فاز 1: Collection (جمع‌آوری)

  • لاگ‌های سیستم (auth.log, syslog)
  • لاگ‌های وب‌سرور (access.log, error.log)
  • فرآیندهای در حال اجرا (ps aux)
  • اتصالات شبکه (netstat -tulpn)
  • کاربران و گروه‌ها (/etc/passwd)
  • Cron jobs (/etc/crontab)
  • فایل‌های جدید (find / -mtime -7)
  • PCAP (tcpdump)

فاز 2: Analysis (تحلیل)

  • یافتن اولین دسترسی (Initial Access)
  • شناسایی ماندگاری (Persistence)
  • شناسایی ارتقای دسترسی
  • شناسایی حرکت جانبی
  • شناسایی خروج داده

فاز 3: Indicators (شاخص‌ها)

  • فرآیندهای عجیب
  • اتصالات مشکوک
  • فایل‌های مخرب
  • Cron/Scheduled Tasks جدید
  • کاربران جدید
  • SSH keys جدید

فاز 4: Timeline (خط زمانی)

  • مرتب‌سازی رویدادها
  • شناسایی نقاط عطف
  • بازسازی زنجیره حمله

فاز 5: Malware Triage

  • strings, file
  • Hash (MD5/SHA256)
  • Static Analysis
  • Dynamic Analysis
  • شناسایی نوع بدافزار

💡 نکات طلایی پیشرفته برای CTF

  1. اولین نقطه نفوذ را پیدا کنید: جایی که اولین Accepted password بعد از Failed password است
  2. Persistence را پیدا کنید: SSH keys, Cron jobs, Services
  3. حرکت جانبی را دنبال کنید: لاگ‌های ماشین‌های دیگر را چک کنید
  4. خروج داده را پیدا کنید: ترافیک خروجی به IP های مخرب
  5. خط زمانی دقیق بسازید: همه چیز را بر اساس زمان مرتب کنید
  6. Web Shell ها را پیدا کنید: فایل‌های PHP با eval, system, shell_exec
  7. SSH Keys را چک کنید: کلیدهای عمومی جدید در authorized_keys
  8. کاربران جدید را چک کنید: کاربران با UID 0 در لینوکس
  9. Scheduled Tasks را چک کنید: کارهای زمان‌بندی شده جدید
  10. لاگ‌های PowerShell را چک کنید: اسکریپت‌های مخرب
