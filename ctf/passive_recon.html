1️⃣ WHOIS - اطلاعات ثبت دامنه

📌 هدف: دریافت اطلاعات کامل ثبت دامنه شامل ثبت‌کننده، تاریخ‌ها، Name Servers و سازمان

Base دستورات پایه

bash 
# اطلاعات کامل دامنه
whois example.com

# فقط اطلاعات خاص با grep
whois example.com | grep -i "registrar"
whois example.com | grep -i "creation date"
whois example.com | grep -i "expiry date"
whois example.com | grep -i "name server"
whois example.com | grep -i "orgname\|organization"

# ذخیره در فایل
whois example.com > whois_output.txt

CTF سناریوهای CTF

bash 
# 1. پیدا کردن تاریخ ثبت دامنه (سوالات CTF)
whois example.com | grep -i "creation date" | awk -F': ' '{print $2}' | cut -d' ' -f1

# 2. پیدا کردن Name Server اصلی
whois example.com | grep -i "name server" | head -1 | awk '{print $NF}'

# 3. بررسی Privacy Protection
if whois example.com | grep -qi "redacted\|privacy\|whoisguard"; then
    echo "[!] WHOIS Privacy is enabled"
else
    echo "[+] WHOIS information is visible"
fi

# 4. پیدا کردن ایمیل از WHOIS
whois example.com | grep -i "email\|@"

# 5. بررسی وضعیت دامنه
whois example.com | grep -i "status" | grep -v "N/A"

Tip ترفندهای پیشرفته

bash 
# استخراج خودکار همه فیلدهای مهم
whois example.com | awk -F': ' '
/Registrar/ {print "Registrar: " $2}
/Creation Date/ {print "Creation: " $2}
/Expiry Date/ {print "Expiry: " $2}
/Name Server/ {print "NS: " $2}
/OrgName/ {print "Org: " $2}
'

# مقایسه دو دامنه
diff <(whois example1.com) <(whois example2.com)

2️⃣ DIG - جستجوی پیشرفته DNS

📌 هدف: دریافت دقیق و کامل رکوردهای DNS با جزئیات بیشتر

Base دستورات پایه

bash 
# رکورد A - آدرس IPv4
dig example.com A +short

# رکورد AAAA - آدرس IPv6
dig example.com AAAA +short

# رکورد MX - سرورهای ایمیل
dig example.com MX +short

# رکورد NS - سرورهای نام
dig example.com NS +short

# رکورد TXT - متون (SPF, DKIM, DMARC)
dig example.com TXT +short

# رکورد CNAME - نام‌های مستعار
dig example.com CNAME +short

# رکورد PTR - Reverse DNS
dig -x 8.8.8.8 +short

# همه رکوردها
dig example.com ANY

CTF سناریوهای CTF

bash 
# 1. پیدا کردن IP وب‌سایت
dig example.com A +short | head -1

# 2. پیدا کردن سرورهای ایمیل
dig example.com MX +short | awk '{print $2}' | head -3

# 3. پیدا کردن SPF Record
dig example.com TXT +short | grep -i "v=spf1"

# 4. پیدا کردن DKIM Record
dig default._domainkey.example.com TXT +short

# 5. پیدا کردن DMARC Record
dig _dmarc.example.com TXT +short

# 6. تست Zone Transfer (اگر باز باشد اطلاعات کاملی می‌دهد)
dig example.com AXFR @ns1.example.com

Tip ترفندهای پیشرفته

bash 
# پیدا کردن زیردامنه‌ها از طریق DNS (Brute Force)
for sub in www mail ftp ssh admin test dev api; do
    dig $sub.example.com A +short | while read ip; do
        if [ ! -z "$ip" ]; then
            echo "[+] $sub.example.com -> $ip"
        fi
    done
done

# مقایسه پاسخ از DNS مختلف
for dns in 8.8.8.8 1.1.1.1 9.9.9.9; do
    echo "=== $dns ==="
    dig @$dns example.com A +short
done

3️⃣ NSLOOKUP - جستجوی DNS

📌 هدف: جستجوی تعاملی و غیرتعاملی DNS

Base دستورات پایه

bash 
# ==================== حالت غیرتعاملی ====================

# جستجوی ساده
nslookup example.com

# جستجوی نوع خاص
nslookup -type=MX example.com
nslookup -type=NS example.com
nslookup -type=TXT example.com
nslookup -type=ANY example.com

# استفاده از سرور DNS خاص
nslookup example.com 8.8.8.8

# Reverse DNS
nslookup 8.8.8.8

CTF سناریوهای CTF

bash 
# 1. پیدا کردن سرورهای ایمیل
nslookup -type=MX example.com | grep "mail exchanger"

# 2. پیدا کردن Name Servers
nslookup -type=NS example.com | grep "nameserver"

# 3. پیدا کردن SPF
nslookup -type=TXT example.com | grep -i "spf"

4️⃣ HOST - جستجوی DNS ساده

📌 هدف: جستجوی سریع و ساده DNS
bash 
# جستجوی A Record
host example.com

# جستجوی نوع خاص
host -t MX example.com
host -t NS example.com
host -t TXT example.com

# Reverse DNS
host 8.8.8.8

# پیدا کردن همه IPها
host example.com | grep "has address" | awk '{print $4}' | sort -u

# ==================== سناریوهای CTF ====================

# 1. پیدا کردن MX
host -t MX example.com | grep "mail" | awk '{print $NF}'

# 2. پیدا کردن TXT
host -t TXT example.com | grep -i "spf\|dkim"

5️⃣ SUBLIST3R - پیدا کردن زیردامنه

📌 هدف: پیدا کردن زیردامنه‌ها با استفاده از موتورهای جستجو

Base دستورات پایه

bash 
# اسکن ساده
sublist3r -d example.com

# ذخیره در فایل
sublist3r -d example.com -o subdomains.txt

# استفاده از همه موتورها
sublist3r -d example.com -b all

# اسکن پورت‌های خاص
sublist3r -d example.com -p 80,443,8080

CTF سناریوهای CTF

bash 
# 1. پیدا کردن زیردامنه‌های ادمین
sublist3r -d example.com | grep -i "admin\|panel\|login\|portal\|dashboard"

# 2. پیدا کردن زیردامنه‌های تست
sublist3r -d example.com | grep -i "test\|dev\|staging\|uat\|qa"

# 3. پیدا کردن زیردامنه‌های API
sublist3r -d example.com | grep -i "api\|rest\|graphql"

# 4. ترکیب با theHarvester
sublist3r -d example.com | while read sub; do
    theHarvester -d $sub -b google 2>/dev/null | grep "@"
done

6️⃣ AMASS - زیردامنه پیشرفته

📌 هدف: پیدا کردن زیردامنه‌ها با استفاده از منابع متعدد و تکنیک‌های پیشرفته
bash 
# اسکن Passive
amass enum -passive -d example.com

# اسکن Active
amass enum -active -d example.com

# ذخیره در فایل
amass enum -d example.com -o amass_output.txt

# ==================== سناریوهای CTF ====================

# 1. پیدا کردن زیردامنه‌های Cloud
amass enum -passive -d example.com | grep -i "cloud\|aws\|azure\|gcp"

# 2. پیدا کردن زیردامنه‌های تست
amass enum -passive -d example.com | grep -i "test\|dev\|staging\|uat"

# 3. ترکیب با Shodan
amass enum -passive -d example.com | head -10 | while read sub; do
    ip=$(dig $sub A +short | head -1)
    if [ -n "$ip" ]; then
        shodan host $ip 2>/dev/null | head -5
    fi
done

7️⃣ DNSRECON - Enumeration کامل DNS

📌 هدف: Enumeration کامل و پیشرفته DNS با تکنیک‌های مختلف
bash 
# اسکن استاندارد
dnsrecon -d example.com -t std

# Zone Transfer
dnsrecon -d example.com -t axfr

# Brute Force
dnsrecon -d example.com -t brt -D /usr/share/wordlists/dnsmap.txt

# ==================== سناریوهای CTF ====================

# 1. پیدا کردن همه IPها
dnsrecon -d example.com -t std | grep "A " | awk '{print $3}'

# 2. بررسی Zone Transfer
if dnsrecon -d example.com -t axfr | grep -q "A "; then
    echo "[!] Zone Transfer is possible!"
else
    echo "[+] Zone Transfer is secure"
fi

8️⃣ THEHARVESTER - جمع‌آوری ایمیل و زیردامنه

📌 هدف: جمع‌آوری ایمیل‌ها، زیردامنه‌ها و اطلاعات از موتورهای جستجو

# جستجو در همه منابع
theHarvester -d example.com -b all

# ذخیره در فایل HTML
theHarvester -d example.com -b google -f output.html

# ==================== سناریوهای CTF ====================

# 1. پیدا کردن ایمیل‌های مهم
theHarvester -d example.com -b all | grep -i "admin\|ceo\|cto\|founder\|info\|support"

# 2. پیدا کردن زیردامنه‌ها
theHarvester -d example.com -b all | grep -o "[a-zA-Z0-9.-]*\.example.com" | sort -u

# 3. استخراج فقط ایمیل‌ها
theHarvester -d example.com -b google | grep "@example.com" | sort -u > emails.txt

9️⃣ SHODAN - جستجوی دستگاه‌های متصل

📌 هدف: جستجوی دستگاه‌ها و سرویس‌های متصل به اینترنت
bash 
# اطلاعات یک IP
shodan host 8.8.8.8

# جستجوی دامنه
shodan search domain:example.com

# ==================== سناریوهای CTF ====================

# 1. پیدا کردن اطلاعات SSL
shodan search "ssl.cert.subject.CN:example.com"

# 2. پیدا کردن آسیب‌پذیری‌ها (مثلا Log4j)
shodan search "vuln:CVE-2021-44228"

# 3. پیدا کردن دستگاه‌های IoT
shodan search "MikroTik"

# 4. پیدا کردن سرویس‌های SSH
shodan search "port:22 product:OpenSSH"

# 5. پیدا کردن دیتابیس‌ها
shodan search "port:3306 product:MySQL"

🔟 NMAP - اسکن پورت و سرویس

📌 هدف: اسکن پورت‌ها، کشف سرویس‌ها و Enumeration

Base دستورات پایه

bash 
# اسکن سریع (100 پورت)
nmap -T4 -F target.com

# اسکن کامل (همه پورت‌ها)
nmap -T4 -p- target.com

# تشخیص سرویس
nmap -sV -p 80,443 target.com

# تشخیص OS
nmap -O target.com

# اسکن با اسکریپت‌ها
nmap -sC target.com

# اسکن UDP
nmap -sU -p 53,161 target.com

CTF سناریوهای CTF

bash 
# 1. اسکن کامل برای CTF
nmap -sC -sV -O -p- -T4 target.com -oA full_scan

# 2. پیدا کردن سرویس‌های خاص
nmap -sV target.com | grep -i "http\|https\|ssh\|ftp\|mysql"

# 3. اسکن سرورهای وب
nmap --script=http-headers target.com
nmap --script=http-enum target.com

# 4. اسکن با اسکریپت‌های آسیب‌پذیری
nmap --script=vuln target.com

# 5. پیدا کردن هاست‌های زنده در شبکه
nmap -sn 192.168.1.0/24 | grep "up"

1️⃣3️⃣ CURL - بررسی وب‌سایت

📝 اسکریپت کامل اتوماسیون (Auto Recon)

📌 هدف: اجرای خودکار همه ابزارها برای جمع‌آوری سریع اطلاعات در CTF
bash 
#!/bin/bash
# ctf_complete_recon.sh

TARGET=${1:-example.com}
OUTPUT_DIR="recon_${TARGET}_$(date +%Y%m%d_%H%M%S)"
mkdir -p $OUTPUT_DIR

RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; CYAN='\033[0;36m'; NC='\033[0m'

echo -e "${CYAN}=========================================${NC}"
echo -e "${YELLOW}   CTF COMPLETE RECON TOOL${NC}"
echo -e "${CYAN}=========================================${NC}"
echo -e "${GREEN}Target: ${TARGET}${NC}"

# 1. WHOIS
echo -e "${YELLOW}[1/10] Running WHOIS...${NC}"
whois $TARGET > $OUTPUT_DIR/01_whois.txt 2>/dev/null

# 2. DIG
echo -e "${YELLOW}[2/10] Running DIG...${NC}"
for type in A AAAA MX NS TXT CNAME SOA; do
    dig $TARGET $type +short > $OUTPUT_DIR/02_dns_${type}.txt 2>/dev/null
done

# 3. Sublist3r
echo -e "${YELLOW}[3/10] Running Sublist3r...${NC}"
sublist3r -d $TARGET -o $OUTPUT_DIR/03_sublist3r.txt &>/dev/null

# 4. Amass
echo -e "${YELLOW}[4/10] Running Amass...${NC}"
amass enum -passive -d $TARGET -o $OUTPUT_DIR/04_amass.txt &>/dev/null

# 5. DNSRecon
echo -e "${YELLOW}[5/10] Running DNSRecon...${NC}"
dnsrecon -d $TARGET -t std > $OUTPUT_DIR/05_dnsrecon.txt 2>/dev/null

# 6. theHarvester
echo -e "${YELLOW}[6/10] Running theHarvester...${NC}"
theHarvester -d $TARGET -b all > $OUTPUT_DIR/06_harvester.txt 2>/dev/null

# 7. Shodan
echo -e "${YELLOW}[7/10] Running Shodan...${NC}"
dig $TARGET A +short | head -3 | while read ip; do
    shodan host $ip > $OUTPUT_DIR/07_shodan_${ip}.txt 2>/dev/null
done

# 8. Nmap
echo -e "${YELLOW}[8/10] Running Nmap...${NC}"
dig $TARGET A +short | head -1 | while read ip; do
    nmap -T4 -F $ip > $OUTPUT_DIR/08_nmap_${ip}.txt 2>/dev/null
done

# 9. Google Dorks File
echo -e "${YELLOW}[9/10] Generating Google Dorks...${NC}"
cat > $OUTPUT_DIR/09_google_dorks.txt << EOF
site:$TARGET intitle:admin
site:$TARGET filetype:pdf
site:$TARGET "password"
EOF

# 10. CURL
echo -e "${YELLOW}[10/10] Running CURL...${NC}"
curl -I -L -s $TARGET > $OUTPUT_DIR/10_headers.txt 2>/dev/null

echo -e "${GREEN}✅ RECON COMPLETE${NC}"
echo -e "${CYAN}Output: $OUTPUT_DIR${NC}"
echo -e "Subdomains: $(cat $OUTPUT_DIR/03_sublist3r.txt 2>/dev/null | wc -l)"
echo -e "Emails: $(grep -c '@' $OUTPUT_DIR/06_harvester.txt 2>/dev/null)"
