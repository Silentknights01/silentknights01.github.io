A️⃣ Network Scanning

1️⃣ NMAP - Host Discovery

📌 هدف: پیدا کردن هاست‌های زنده در شبکه

Base دستورات پایه

bash 
# ==================== Ping Sweep ====================

# ICMP Ping Sweep
nmap -sn 192.168.1.0/24

# TCP SYN Ping
nmap -sn -PS 80 192.168.1.0/24

# TCP ACK Ping
nmap -sn -PA 80 192.168.1.0/24

# UDP Ping
nmap -sn -PU 53 192.168.1.0/24

# ARP Ping (Local Network)
nmap -sn -PR 192.168.1.0/24

# ==================== دستورات ترکیبی ====================

# ترکیب چند روش
nmap -sn -PS 80 -PA 22 -PU 53 192.168.1.0/24

# بدون Ping (فرض کن زنده است)
nmap -Pn 192.168.1.10

# با لیست IP
nmap -sn 192.168.1.1-254

# با CIDR
nmap -sn 192.168.1.0/24

CTF سناریوهای CTF

bash 
# 1. پیدا کردن همه هاست‌های زنده
nmap -sn 192.168.1.0/24 | grep "Host is up" -B 2 | grep "Nmap scan" | awk '{print $5}'

# 2. پیدا کردن هاست‌های با پورت خاص
nmap -sn -PS 22 192.168.1.0/24

# 3. پیدا کردن هاست‌های با ARP
nmap -sn -PR 192.168.1.0/24

# 4. پیدا کردن فقط هاست‌های با پاسخ
nmap -sn 192.168.1.0/24 | grep -B 2 "Host is up" | grep "Nmap scan" | awk '{print $5}'

# 5. پیدا کردن هاست‌های با IP Range
nmap -sn 192.168.1.1-50

# 6. پیدا کردن هاست‌های با CIDR
nmap -sn 192.168.1.0/24

# 7. پیدا کردن هاست‌های با TCP SYN
nmap -sn -PS 80,443 192.168.1.0/24

# 8. پیدا کردن هاست‌های با UDP
nmap -sn -PU 53,161 192.168.1.0/24

# 9. پیدا کردن هاست‌های با ICMP
nmap -sn -PE 192.168.1.0/24

# 10. پیدا کردن هاست‌های با Timestamp
nmap -sn -PP 192.168.1.0/24

# 11. پیدا کردن هاست‌های با Subnet Mask
nmap -sn -PM 192.168.1.0/24

# 12. پیدا کردن هاست‌های با SCTP
nmap -sn -PY 80 192.168.1.0/24

# 13. پیدا کردن هاست‌های با همه روش‌ها
nmap -sn -PE -PP -PM -PS 80 -PA 22 -PU 53 192.168.1.0/24

# 14. پیدا کردن هاست‌های با خروجی
nmap -sn -oA hosts 192.168.1.0/24

# 15. پیدا کردن هاست‌های با نمایش IP
nmap -sn 192.168.1.0/24 | grep -o '[0-9]\{1,3\}\.[0-9]\{1,3\}\.[0-9]\{1,3\}\.[0-9]\{1,3\}'

Tip نکات مهم CTF

bash 
# 1. Ping Sweep با fping
fping -a -g 192.168.1.0/24

# 2. Ping Sweep با hping3
hping3 -S -p 80 --scan 1-1000 192.168.1.0/24

# 3. Ping Sweep با bash
for i in {1..254}; do ping -c 1 -W 1 192.168.1.$i | grep "64 bytes" & done

# 4. Ping Sweep با parallel
seq 1 254 | parallel -j 50 "ping -c 1 -W 1 192.168.1.{} | grep '64 bytes'"

# 5. Ping Sweep با nmap و خروجی
nmap -sn 192.168.1.0/24 -oG - | awk '/Up$/{print $2}'

2️⃣ ARP-SCAN & 3️⃣ NETDISCOVER

bash 
# ==================== ARP-SCAN ====================

# اسکن کامل شبکه
arp-scan 192.168.1.0/24

# اسکن با Interface
arp-scan -I eth0 192.168.1.0/24

# اسکن با تاخیر
arp-scan -t 1000 192.168.1.0/24

# اسکن با نمایش فقط IP
arp-scan 192.168.1.0/24 | grep -o '[0-9]\{1,3\}\.[0-9]\{1,3\}\.[0-9]\{1,3\}\.[0-9]\{1,3\}'

# اسکن با نمایش MAC
arp-scan 192.168.1.0/24 | grep -o '[a-fA-F0-9:]{17}'

# اسکن با خروجی
arp-scan 192.168.1.0/24 > arp_scan.txt

# ==================== NETDISCOVER ====================

# اسکن شبکه
netdiscover -r 192.168.1.0/24

# اسکن با Interface
netdiscover -i eth0 -r 192.168.1.0/24

# اسکن Passive
netdiscover -p

# اسکن با خروجی
netdiscover -r 192.168.1.0/24 -o netdiscover.txt

# اسکن با Timeout
netdiscover -r 192.168.1.0/24 -t 100

# اسکن Passive با N
netdiscover -r 192.168.1.0/24 -N

# اسکن با S
netdiscover -r 192.168.1.0/24 -S

B️⃣ Port & Service Scanning

4️⃣ NMAP - Port Scanning

📌 هدف: پیدا کردن پورت‌های باز
bash 
# ==================== تکنیک‌های اسکن ====================

# SYN Scan (نیمه‌باز - سریع)
nmap -sS 192.168.1.10

# TCP Connect Scan (کامل)
nmap -sT 192.168.1.10

# UDP Scan
nmap -sU 192.168.1.10

# FIN Scan (Stealth)
nmap -sF 192.168.1.10

# NULL Scan
nmap -sN 192.168.1.10

# XMAS Scan
nmap -sX 192.168.1.10

# ACK Scan (Firewall Detection)
nmap -sA 192.168.1.10

# Window Scan
nmap -sW 192.168.1.10

# Maimon Scan
nmap -sM 192.168.1.10

# ==================== دستورات ترکیبی ====================

# اسکن سریع (پورت‌های Top 100)
nmap -T4 -F 192.168.1.10

# اسکن پورت‌های خاص
nmap -p 22,80,443 192.168.1.10

# اسکن رنج پورت
nmap -p 1-1000 192.168.1.10

# اسکن همه پورت‌ها
nmap -p- 192.168.1.10

# اسکن با محدودیت زمان
nmap -T5 -p- --max-rtt-timeout 1000ms 192.168.1.10

# ==================== سناریوهای CTF ====================

# 1. اسکن سریع پورت‌های رایج
nmap -T4 -F 192.168.1.10

# 2. اسکن همه پورت‌ها
nmap -p- -T4 192.168.1.10

# 3. اسکن پورت‌های خاص
nmap -p 21,22,25,53,80,110,111,135,139,143,443,445,993,995,1723,3306,3389,5900,8080 192.168.1.10

# 4. اسکن با مشخص کردن پورت‌ها از فایل
nmap -p $(cat ports.txt | tr '\n' ',') 192.168.1.10

# 5. اسکن فقط پورت‌های باز
nmap -p- --open 192.168.1.10

# 6. اسکن با تحمل خطا
nmap -p- --max-retries 2 192.168.1.10

# 7. اسکن با تاخیر
nmap -p- --min-rate 1000 192.168.1.10

# 8. اسکن UDP روی پورت‌های رایج
nmap -sU -p 53,161,137,138,139 192.168.1.10

# 9. اسکن با پورت‌های تصادفی
nmap -p 80 --randomize-hosts 192.168.1.10

# 10. اسکن با پورت‌های Top 1000
nmap --top-ports 1000 192.168.1.10

# 20. اسکن با همه تکنیک‌ها
nmap -sS -sT -sU -p- 192.168.1.10

Tip ترفندهای پیشرفته Port Scanning

bash 
# 1. اسکن با Fragment (Evasion)
nmap -f -p- 192.168.1.10

# 2. اسکن با MTU
nmap --mtu 24 -p- 192.168.1.10

# 3. اسکن با Decoy
nmap -D RND:10 -p- 192.168.1.10

# 4. اسکن با Source Port
nmap --source-port 53 -p- 192.168.1.10

# 5. اسکن با Data Length
nmap --data-length 200 -p- 192.168.1.10

# 6. اسکن با Spoof MAC
nmap --spoof-mac 00:11:22:33:44:55 -p- 192.168.1.10

# 7. اسکن با Bad Checksum
nmap --badsum -p- 192.168.1.10

# 8. اسکن با TTL
nmap --ttl 64 -p- 192.168.1.10

# 9. اسکن با Proxy
nmap --proxies http://127.0.0.1:8080 -p- 192.168.1.10

# 10. اسکن با All Ports + Service
nmap -sS -sV -p- -T4 192.168.1.10

5️⃣ NMAP - Service Detection

📌 هدف: تشخیص سرویس‌ها و نسخه‌ها
bash 
# ==================== دستورات اصلی ====================

# تشخیص سرویس
nmap -sV 192.168.1.10

# تشخیص سرویس با شدت بالا
nmap -sV --version-intensity 9 192.168.1.10

# تشخیص سرویس با نسخه
nmap -sV --version-all 192.168.1.10

# تشخیص سرویس روی پورت خاص
nmap -sV -p 80 192.168.1.10

# ==================== دستورات ترکیبی ====================

# تشخیص سرویس و اسکن همه پورت‌ها
nmap -sV -p- 192.168.1.10

# تشخیص سرویس با اسکریپت‌ها
nmap -sV -sC 192.168.1.10

# تشخیص سرویس با نسخه و OS
nmap -sV -O 192.168.1.10

# ==================== سناریوهای CTF ====================

# 1. تشخیص سرویس روی پورت‌های باز
nmap -sV -p 22,80,443 192.168.1.10

# 2. تشخیص سرویس با شدت کم
nmap -sV --version-intensity 0 192.168.1.10

# 3. تشخیص سرویس با شدت بالا
nmap -sV --version-intensity 9 --version-all 192.168.1.10

# 4. تشخیص سرویس و ذخیره خروجی
nmap -sV -oN service_scan.txt 192.168.1.10

# 5. تشخیص سرویس با XML خروجی
nmap -sV -oX service_scan.xml 192.168.1.10

# 6. تشخیص سرویس فقط روی TCP
nmap -sV -sT 192.168.1.10

# 7. تشخیص سرویس با timeout بیشتر
nmap -sV --host-timeout 5m 192.168.1.10

# 8. تشخیص سرویس با max-retries
nmap -sV --max-retries 5 192.168.1.10

# 9. تشخیص سرویس با Light
nmap -sV --version-light 192.168.1.10

# 10. تشخیص سرویس با Heavy
nmap -sV --version-heavy 192.168.1.10

# 11. تشخیص سرویس با TCP
nmap -sV -sS 192.168.1.10

# 12. تشخیص سرویس با UDP
nmap -sV -sU 192.168.1.10

# 13. تشخیص سرویس با همه پورت‌ها
nmap -sV -p- -T4 192.168.1.10

# 14. تشخیص سرویس با خروجی Grep
nmap -sV -oG service.grep 192.168.1.10

# 15. تشخیص سرویس با خروجی JSON
nmap -sV -oJ service.json 192.168.1.10

6️⃣ NMAP - OS Detection

📌 هدف: تشخیص سیستم‌عامل
bash 
# ==================== دستورات اصلی ====================

# تشخیص OS
nmap -O 192.168.1.10

# تشخیص OS با حدس دقیق
nmap -O --osscan-guess 192.168.1.10

# تشخیص OS با نشان دادن احتمالات
nmap -O --osscan-limit 192.168.1.10

# ==================== دستورات ترکیبی ====================

# تشخیص OS و سرویس
nmap -O -sV 192.168.1.10

# تشخیص OS با اسکن همه پورت‌ها
nmap -O -p- 192.168.1.10

# ==================== سناریوهای CTF ====================

# 1. تشخیص OS
nmap -O 192.168.1.10 | grep "OS details"

# 2. تشخیص OS با دقت بالا
nmap -O --osscan-guess --fuzzy 192.168.1.10

# 3. تشخیص OS با اسکن پورت‌های رایج
nmap -O -F 192.168.1.10

# 4. تشخیص OS با نشان دادن همه احتمالات
nmap -O --osscan-limit --osscan-guess 192.168.1.10

# 5. تشخیص OS و ذخیره خروجی
nmap -O -oN os_scan.txt 192.168.1.10

# 6. تشخیص OS از روی TTL
ping -c 1 192.168.1.10 | grep -o "ttl=[0-9]*" | cut -d'=' -f2
# 64 = Linux, 128 = Windows

# 7. تشخیص OS با TCP/IP Fingerprint
nmap -O --osscan-guess 192.168.1.10

# 8. تشخیص OS با Aggressive
nmap -O -A 192.168.1.10

# 9. تشخیص OS با Scripts
nmap -O --script=* 192.168.1.10

# 10. تشخیص OS با خروجی
nmap -O -oA os_scan 192.168.1.10

📝 اسکریپت کامل Scanning & Vulnerability Auto

📌 هدف: اجرای خودکار همه ابزارها برای جمع‌آوری سریع اطلاعات
bash 
#!/bin/bash
# scan_vuln_cheatsheet.sh - اسکریپت کامل Scanning & Vulnerability

TARGET=${1:-192.168.1.10}
OUTPUT_DIR="scan_vuln_$(date +%Y%m%d_%H%M%S)"
mkdir -p $OUTPUT_DIR

echo "========================================="
echo "  SCANNING & VULNERABILITY CHEAT SHEET"
echo "========================================="

# A. Network Scanning
echo "[A1] Running Network Scan..."
nmap -sn $TARGET/24 > $OUTPUT_DIR/A1_network_scan.txt
echo "✓ Network Scan saved"

# B. Port Scanning
echo "[B1] Running Basic Port Scan..."
nmap -T4 -F $TARGET > $OUTPUT_DIR/B1_ports_basic.txt
echo "✓ Basic Port Scan saved"

echo "[B2] Running Full Port Scan..."
nmap -p- -T4 $TARGET > $OUTPUT_DIR/B2_ports_full.txt
echo "✓ Full Port Scan saved"

# C. Service Detection
echo "[C1] Running Service Detection..."
nmap -sV -p 21,22,25,53,80,110,111,135,139,143,443,445,993,995,1723,3306,3389,5900,8080 $TARGET > $OUTPUT_DIR/C1_services.txt
echo "✓ Service Detection saved"

echo "[C2] Running OS Detection..."
nmap -O $TARGET > $OUTPUT_DIR/C2_os.txt
echo "✓ OS Detection saved"

# D. Vulnerability Scanning
echo "[D1] Running Nmap NSE Vulnerability..."
nmap --script=vuln -p 80,443 $TARGET > $OUTPUT_DIR/D1_nmap_nse.txt
echo "✓ Nmap NSE saved"

echo "[D2] Running Nikto..."
if command -v nikto &> /dev/null; then
    nikto -h $TARGET -ssl -o $OUTPUT_DIR/D2_nikto.html > $OUTPUT_DIR/D2_nikto.txt
    echo "✓ Nikto saved"
else
    echo "✗ Nikto not installed"
fi

echo "[D3] Running Nuclei..."
if command -v nuclei &> /dev/null; then
    nuclei -u $TARGET -severity critical,high -o $OUTPUT_DIR/D3_nuclei.txt
    echo "✓ Nuclei saved"
else
    echo "✗ Nuclei not installed"
fi

echo "[D4] Running SearchSploit..."
searchsploit $TARGET > $OUTPUT_DIR/D4_searchsploit.txt
echo "✓ SearchSploit saved"

# E. Web Analysis
echo "[E1] Running WhatWeb..."
whatweb -a 3 $TARGET > $OUTPUT_DIR/E1_whatweb.txt
echo "✓ WhatWeb saved"

echo "[E2] Running Wafw00f..."
wafw00f $TARGET > $OUTPUT_DIR/E2_wafw00f.txt
echo "✓ Wafw00f saved"

echo ""
echo "========================================="
echo "  ✅ SCAN COMPLETE"
echo "========================================="
echo "📁 Output: $OUTPUT_DIR"
ls -la $OUTPUT_DIR
