1️⃣ FTP Enumeration (Port 21)

📌 هدف: Enumeration سرویس FTP برای پیدا کردن اطلاعات، فایل‌ها و دسترسی‌ها

Base دستورات پایه

bash 
# ==================== Nmap Scripts ====================

# اسکن FTP با Nmap
nmap -sV -p21 --script=ftp-* TARGET

# اسکن با تشخیص نسخه
nmap -sV -p21 --script=ftp-anon,ftp-bounce,ftp-proftpd-backdoor TARGET

# اسکن کامل FTP
nmap -p21 --script=ftp-anon,ftp-bounce,ftp-brute,ftp-proftpd-backdoor,ftp-syst TARGET

# ==================== Anonymous Login ====================

# اتصال با Anonymous
ftp TARGET
> Username: anonymous
> Password: anonymous

# اتصال با ftp command
ftp -n TARGET <

CTF سناریوهای CTF

bash 
# 1. بررسی Anonymous
ftp -n TARGET <

Tip نکات مهم CTF

bash 
# 1. اسکن کامل FTP
nmap -p21 --script=ftp-* -sV TARGET

# 2. دانلود recursive
wget -r ftp://anonymous:anonymous@TARGET/

# 3. بررسی فایل‌های مخفی
ftp -n TARGET <

2️⃣ SSH Enumeration (Port 22)

📌 هدف: Enumeration سرویس SSH برای پیدا کردن نسخه و کاربران

Base دستورات پایه

bash 
# ==================== Nmap Scripts ====================

# اسکن SSH با Nmap
nmap -sV -p22 --script=ssh-* TARGET

# اسکن با تشخیص نسخه
nmap -sV -p22 --script=ssh-hostkey,ssh-auth-methods TARGET

# اسکن کامل SSH
nmap -p22 --script=ssh-* -sV TARGET

# ==================== Version & Banner ====================

# دریافت نسخه
nmap -sV -p22 TARGET | grep ssh

# دریافت بنر
nc -nv TARGET 22

# ==================== Brute Force ====================

# Hydra SSH Brute Force
hydra -l root -P /usr/share/wordlists/rockyou.txt ssh://TARGET -t 4 -f

# Hydra با لیست کاربران
hydra -L users.txt -P /usr/share/wordlists/rockyou.txt ssh://TARGET -t 4

CTF سناریوهای CTF

bash 
# 1. بررسی نسخه
nmap -sV -p22 TARGET | grep -i "ssh"

# 2. دریافت بنر
nc -nv TARGET 22

# 3. بررسی روش‌های احراز
nmap -p22 --script=ssh-auth-methods TARGET

# 4. بررسی کلیدهای میزبان
nmap -p22 --script=ssh-hostkey TARGET

# 5. بررسی SSHv1
nmap -p22 --script=sshv1 TARGET

# 6. بررسی با hydra
hydra -l admin -P /usr/share/wordlists/rockyou.txt ssh://TARGET -t 4 -vV

# 7. اتصال با ssh
ssh -o StrictHostKeyChecking=no user@TARGET

# 8. بررسی پورت های جایگزین
nmap -p22,2222,22222 --script=ssh-* TARGET

# 9. بررسی RSA key
ssh -v user@TARGET 2>&1 | grep -i "key"

# 10. بررسی با medusa
medusa -h TARGET -u root -P /usr/share/wordlists/rockyou.txt -M ssh -t 4

Tip نکات مهم CTF

bash 
# 1. اسکن کامل SSH
nmap -p22 --script=ssh-* -sV -A TARGET

# 2. بررسی CVE
nmap -p22 --script=ssh-* --script-args=ssh.vulncheck=TARGET

# 3. بررسی با SSH Weak Ciphers
nmap -p22 --script=ssh2-enum-algos TARGET

# 4. بررسی با SSH1
nmap -p22 --script=sshv1 TARGET

# 5. بررسی با hydra با کاربران مختلف
hydra -L /usr/share/wordlists/metasploit/unix_users.txt -P /usr/share/wordlists/rockyou.txt ssh://TARGET -t 4

4️⃣ DNS Enumeration (Port 53)

📌 هدف: Enumeration سرویس DNS برای پیدا کردن زیردامنه‌ها و اطلاعات

Base دستورات پایه

bash 
# ==================== Nmap Scripts ====================

# اسکن DNS با Nmap
nmap -sV -p53 --script=dns-* TARGET

# اسکن با Zone Transfer
nmap -p53 --script=dns-zone-transfer TARGET

# ==================== Zone Transfer ====================

# بررسی Zone Transfer
dnsrecon -d example.com -t axfr

# با dig
dig @TARGET example.com AXFR

# با host
host -t AXFR example.com TARGET

# ==================== Subdomain Enumeration ====================

# با dnsrecon
dnsrecon -d example.com -t std
dnsrecon -d example.com -t brt
dnsrecon -d example.com -t brt -D /usr/share/wordlists/dnsmap.txt

# با dnsenum
dnsenum example.com
dnsenum -f /usr/share/wordlists/dnsmap.txt example.com

# با dig
for sub in www mail ftp ssh admin test dev api; do
    dig $sub.example.com A +short
done

CTF سناریوهای CTF

bash 
# 1. Zone Transfer
dnsrecon -d example.com -t axfr

# 2. Zone Transfer با dig
dig @TARGET example.com AXFR

# 3. Zone Transfer با host
host -t AXFR example.com TARGET

# 4. Subdomain Enumeration
dnsrecon -d example.com -t std

# 5. Subdomain Brute Force
dnsrecon -d example.com -t brt -D /usr/share/wordlists/dnsmap.txt

# 6. با dnsenum
dnsenum example.com -o dnsenum.txt

# 7. با dnsrecon روی زیردامنه‌ها
dnsrecon -d example.com -t brt -D /usr/share/wordlists/dirb/common.txt

# 8. با Nmap Zone Transfer
nmap -p53 --script=dns-zone-transfer TARGET

# 9. با Nmap DNS Recursion
nmap -p53 --script=dns-recursion TARGET

# 10. با Nmap DNS NSEC
nmap -p53 --script=dns-nsec-enum TARGET

# 11. با Nmap DNS Service
nmap -p53 --script=dns-service TARGET

# 12. با dnsrecon روی همه موارد
dnsrecon -d example.com -t all

# 13. با dig روی زیردامنه‌ها
for word in $(cat /usr/share/wordlists/dnsmap.txt); do
    dig $word.example.com A +short
done

# 14. با fuzz
ffuf -u http://example.com -H "Host: FUZZ.example.com" -w /usr/share/wordlists/dnsmap.txt

# 15. بررسی کامل
dnsrecon -d example.com -t all -o dns_output.txt

Tip نکات مهم CTF

bash 
# 1. بررسی Wildcard
dig *.example.com A +short

# 2. بررسی DNSSEC
dig example.com A +dnssec

dnsrecon -d example.com -t nsec

# 4. بررسی با dnsrecon روی SRV
dnsrecon -d example.com -t srv

# 5. Reverse DNS
dnsrecon -r 192.168.1.0/24

# 6. بررسی با Nmap
nmap -p53 --script=dns-* -sU TARGET

# 7. بررسی با dig و لیست
dig example.com +noall +answer +norecurse NS

# 8. بررسی با dig و trace
dig example.com +trace

5️⃣ SMB Enumeration (Port 139/445) ⭐⭐⭐⭐⭐

📌 هدف: Enumeration سرویس SMB برای پیدا کردن Shares, Users, Permissions

Base دستورات پایه

bash 
# ==================== Nmap Scripts ====================

# اسکن SMB با Nmap
nmap -p139,445 --script=smb-* TARGET

# اسکن با Enum Shares
nmap -p139,445 --script=smb-enum-shares TARGET

# اسکن با Enum Users
nmap -p139,445 --script=smb-enum-users TARGET

# اسکن کامل
nmap -p139,445 --script=smb-* -sV TARGET

# ==================== enum4linux ====================

# Enumeration کامل
enum4linux TARGET

# Enumeration Users
enum4linux -U TARGET

# Enumeration Shares
enum4linux -S TARGET

# Enumeration OS
enum4linux -o TARGET

# ==================== smbclient ====================

# لیست Shares
smbclient -L //TARGET -N

# اتصال به Share
smbclient //TARGET/SHARE -N

# اتصال با Credentials
smbclient //TARGET/SHARE -U username

# ==================== smbmap ====================

# بررسی Shares
smbmap -H TARGET

# بررسی با Credentials
smbmap -H TARGET -u username -p password

CTF سناریوهای CTF

bash 
# 1. enum4linux کامل
enum4linux TARGET > enum4linux.txt

# 2. enum4linux Users
enum4linux -U TARGET

# 3. enum4linux Shares
enum4linux -S TARGET

# 4. enum4linux OS
enum4linux -o TARGET

# 5. smbclient لیست
smbclient -L //TARGET -N

# 6. smbclient اتصال
smbclient //TARGET/Public -N

# 7. smbclient با Credentials
smbclient //TARGET/Backup -U admin%password

# 8. smbmap
smbmap -H TARGET

# 9. smbmap با Credentials
smbmap -H TARGET -u admin -p password

# 10. Nmap SMB Enum
nmap -p445 --script=smb-enum-shares TARGET

# 11. Nmap SMB Users
nmap -p445 --script=smb-enum-users TARGET

# 12. Nmap SMB OS
nmap -p445 --script=smb-os-discovery TARGET

# 13. Nmap SMB Security
nmap -p445 --script=smb-security-mode TARGET

# 14. Nmap SMB Vuln
nmap -p445 --script=smb-vuln-* TARGET

# 15. بررسی با rpcclient
rpcclient -U "" -N TARGET
> srvinfo
> enumdomusers
> enumdomgroups
> shares
> quit

# 16. دانلود از SMB
smbclient //TARGET/Public -N -c "get file.txt"

# 17. آپلود به SMB
smbclient //TARGET/Public -N -c "put local.txt"

# 18. بررسی با smbget
smbget -R smb://TARGET/Public

# 19. بررسی با smbtree
smbtree -N

# 20. بررسی کامل SMB
nmap -p139,445 --script=smb-* -sV TARGET -oA smb_scan

Tip نکات مهم CTF

bash 
# 1. بررسی SMB Signing
nmap -p445 --script=smb-security-mode TARGET

# 2. بررسی SMB1
nmap -p445 --script=smb-protocols TARGET

# 3. بررسی Null Session
enum4linux -a TARGET | grep -i "null"

# 4. بررسی با enum4linux روی SMB1
enum4linux -a -u "" -p "" TARGET

# 5. بررسی با smbclient روی همه Shares
for share in $(smbclient -L //TARGET -N 2>/dev/null | grep -E "^[[:space:]]+\w+" | awk '{print $1}'); do
    echo "Testing $share..."
    smbclient //TARGET/$share -N -c "ls" 2>/dev/null
done

# 6. بررسی با Nmap SMB OS
nmap -p445 --script=smb-os-discovery --script-args=smb-os-discovery.use-remote-os TARGET

# 7. بررسی با Nmap SMB Vuln
nmap -p445 --script=smb-vuln-* --script-args=unsafe=1 TARGET

# 8. بررسی با Nmap SMB Enum
nmap -p445 --script=smb-enum-shares,smb-enum-users TARGET

# 9. بررسی با rpcclient
echo "enumdomusers" | rpcclient -U "" -N TARGET

# 10. بررسی با enum4linux تمام اطلاعات
enum4linux -a TARGET 2>/dev/null | tee enum4linux_full.txt

7️⃣ LDAP Enumeration (Port 389)

📌 هدف: Enumeration سرویس LDAP برای پیدا کردن اطلاعات دایرکتوری

Base دستورات پایه

bash 
# ==================== Nmap Scripts ====================

# اسکن LDAP با Nmap
nmap -p389 --script=ldap-* TARGET

# ==================== ldapsearch ====================

# جستجوی پایه
ldapsearch -x -H ldap://TARGET -b ""

# جستجوی Domain
ldapsearch -x -H ldap://TARGET -b "dc=example,dc=com"

# جستجوی Users
ldapsearch -x -H ldap://TARGET -b "dc=example,dc=com" "(objectClass=user)"

CTF سناریوهای CTF

bash

ldapsearch -x -H ldap://TARGET -b ""

ldapsearch -x -H ldap://TARGET -b "dc=example,dc=com" "(objectClass=user)"

# 4. ldapsearch Groups
ldapsearch -x -H ldap://TARGET -b "dc=example,dc=com" "(objectClass=group)"

ldapsearch -x -H ldap://TARGET -b "dc=example,dc=com"

ldapsearch -x -H ldap://TARGET -D "cn=admin,dc=example,dc=com" -w password -b "dc=example,dc=com"

nmap -p389 --script=ldap-* TARGET

# 8. Nmap LDAP RootDSE
nmap -p389 --script=ldap-rootdse TARGET

# 9. Nmap LDAP Search

# 10. دریافت Domain
ldapsearch -x -H ldap://TARGET -b "" | grep -i "domain"

#!/bin/bash
# service_cheatsheet.sh - اسکریپت کامل Service Enumeration

TARGET=${1:-192.168.1.10}
OUTPUT_DIR="service_enum_$(date +%Y%m%d_%H%M%S)"
mkdir -p $OUTPUT_DIR

echo "========================================="
echo "  SERVICE ENUMERATION CHEAT SHEET"
echo "========================================="

# 1. Nmap NSE All
echo "[1/12] Running Nmap NSE..."
nmap -sC -sV $TARGET > $OUTPUT_DIR/01_nmap_nse.txt
echo "✓ Nmap NSE saved"

# 2. FTP
echo "[2/12] Enumerating FTP..."
nmap -p21 --script=ftp-* $TARGET > $OUTPUT_DIR/02_ftp.txt
echo "✓ FTP saved"

# 3. SSH
echo "[3/12] Enumerating SSH..."
nmap -p22 --script=ssh-* $TARGET > $OUTPUT_DIR/03_ssh.txt
echo "✓ SSH saved"

# 4. SMTP
echo "[4/12] Enumerating SMTP..."
nmap -p25 --script=smtp-* $TARGET > $OUTPUT_DIR/04_smtp.txt
echo "✓ SMTP saved"

# 5. DNS
echo "[5/12] Enumerating DNS..."
dnsrecon -d $TARGET -t all > $OUTPUT_DIR/05_dns.txt
echo "✓ DNS saved"

# 6. SMB
echo "[6/12] Enumerating SMB..."
nmap -p139,445 --script=smb-* $TARGET > $OUTPUT_DIR/06_smb.txt
enum4linux $TARGET > $OUTPUT_DIR/06_enum4linux.txt
smbclient -L //$TARGET -N > $OUTPUT_DIR/06_smbclient.txt 2>/dev/null
echo "✓ SMB saved"

# 7. SNMP
echo "[7/12] Enumerating SNMP..."
nmap -sU -p161 --script=snmp-* $TARGET > $OUTPUT_DIR/07_snmp.txt
snmpwalk -v2c -c public $TARGET > $OUTPUT_DIR/07_snmpwalk.txt 2>/dev/null
echo "✓ SNMP saved"

# 8. LDAP
echo "[8/12] Enumerating LDAP..."
nmap -p389 --script=ldap-* $TARGET > $OUTPUT_DIR/08_ldap.txt
echo "✓ LDAP saved"

nmap -p3306 --script=mysql-* $TARGET > $OUTPUT_DIR/09_mysql.txt
echo "✓ MySQL saved"

nmap -p6379 --script=redis-* $TARGET > $OUTPUT_DIR/10_redis.txt
