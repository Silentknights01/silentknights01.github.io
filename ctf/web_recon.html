1️⃣ WHATWEB - Technology Fingerprinting

📌 هدف: شناسایی تکنولوژی‌های استفاده شده در وب‌سایت

Base دستورات پایه

bash 
# اسکن ساده
whatweb example.com

# اسکن با سطح 3 (کامل‌تر)
whatweb -a 3 example.com

# اسکن با سطح 4 (همه چیز)
whatweb -a 4 example.com

# اسکن با نمایش همه چیز
whatweb -v example.com

# اسکن با خروجی JSON
whatweb -a 3 -j example.com

# اسکن با خروجی XML
whatweb -a 3 -x example.com

Combined دستورات ترکیبی

bash 
# اسکن با User-Agent خاص
whatweb -a 3 -U "Mozilla/5.0" example.com

# اسکن با Cookie
whatweb -a 3 -C "PHPSESSID=value" example.com

# اسکن با Proxy
whatweb -a 3 -P "http://127.0.0.1:8080" example.com

# اسکن با Header خاص
whatweb -a 3 -H "X-Forwarded-For: 127.0.0.1" example.com

# اسکن با Timeout
whatweb -a 3 -t 10 example.com

# اسکن با تعداد Thread
whatweb -a 3 -T 10 example.com

CTF سناریوهای CTF

bash 
# 1. دریافت تکنولوژی‌های اصلی
whatweb -a 3 example.com | grep -oE "(WordPress|Apache|PHP|Nginx|jQuery|Bootstrap)"

# 2. دریافت CMS
whatweb -a 3 example.com | grep -i "wordpress\|joomla\|drupal\|magento"

# 3. دریافت Web Server
whatweb -a 3 example.com | grep -i "apache\|nginx\|iis\|lighttpd"

# 4. دریافت Framework
whatweb -a 3 example.com | grep -i "laravel\|django\|rails\|spring"

# 5. دریافت JS Libraries
whatweb -a 3 example.com | grep -i "jquery\|react\|angular\|vue"

# 6. دریافت نسخه‌ها
whatweb -a 3 example.com | grep -oE "[0-9]+\.[0-9]+(?:\.[0-9]+)?"

# 7. دریافت همه اطلاعات در JSON
whatweb -a 3 -j example.com | jq '.'

# 8. اسکن چندین دامنه
for domain in example.com test.com target.org; do
    whatweb -a 3 $domain >> results.txt
done

# 9. اسکن با نمایش فقط تکنولوژی
whatweb -a 3 -q example.com

# 10. اسکن با نمایش بنر
whatweb -a 3 -b example.com

# 11. دریافت IP و تکنولوژی
whatweb -a 3 example.com | grep -o '[0-9]\{1,3\}\.[0-9]\{1,3\}\.[0-9]\{1,3\}\.[0-9]\{1,3\}'

# 12. اسکن با استفاده از HTTPS
whatweb -a 3 https://example.com

Tip نکات مهم CTF

bash 
# 1. تشخیص دقیق CMS
whatweb -a 3 example.com | grep -iE "wordpress|joomla|drupal|magento|shopify|wix"

# 2. تشخیص CDN
whatweb -a 3 example.com | grep -iE "cloudflare|akamai|fastly|cloudfront"

# 3. تشخیص WAF
whatweb -a 3 example.com | grep -iE "cloudflare|sucuri|modsecurity|aws"

# 4. تشخیص زبان برنامه‌نویسی

# 5. تشخیص دیتابیس
whatweb -a 3 example.com | grep -iE "mysql|postgresql|mongodb|mariadb"

# 6. تشخیص Analytics
whatweb -a 3 example.com | grep -iE "google-analytics|gtag|ga.js"

# 7. تشخیص Fonts
whatweb -a 3 example.com | grep -iE "google-fonts|font-awesome"

# 8. اسکن با سرعت بالا
whatweb -a 3 -T 20 example.com

# 9. اسکن با اجتناب از WAF
whatweb -a 3 --random-ua example.com

# 10. اسکن با فیلتر کردن
whatweb -a 3 example.com | grep -v "HTTP"

2️⃣ WAFW00F - WAF Detection

📌 هدف: تشخیص وجود Web Application Firewall

Base دستورات پایه و ترکیبی

bash 
# ==================== دستورات اصلی ====================

# تشخیص WAF
wafw00f example.com

# تشخیص با خروجی JSON
wafw00f -j example.com

# تشخیص با خروجی XML
wafw00f -x example.com

# تشخیص با گزارش
wafw00f -r report.html example.com

# ==================== دستورات ترکیبی ====================

# تشخیص با User-Agent خاص
wafw00f -H "User-Agent: Mozilla/5.0" example.com

# تشخیص با Proxy
wafw00f -p http://127.0.0.1:8080 example.com

# تشخیص چندین دامنه
for domain in example.com test.com; do
    wafw00f $domain
done

CTF سناریوهای CTF

bash 
# 1. تشخیص WAF (نمایش خلاصه)
wafw00f example.com | grep -i "detected\|cloudflare\|sucuri"

# 2. تشخیص دقیق WAF
wafw00f example.com | grep -B 2 -A 2 "WAF"

# 3. تشخیص با خروجی JSON
wafw00f -j example.com | jq '.'

# 4. تشخیص با خروجی XML
wafw00f -x example.com | grep -i "waf"

# 5. تشخیص چندین WAF
wafw00f example.com | grep -i "Cloudflare\|AWS\|Sucuri\|ModSecurity"

# 6. تشخیص با نمایش دلیل
wafw00f -v example.com

# 7. تشخیص با تست کامل
wafw00f -o waf_results.txt example.com

# 8. تشخیص با timeout
wafw00f -t 10 example.com

# 9. تشخیص با همه روش‌ها
wafw00f -a example.com

# 10. تشخیص با فهرست WAFها
wafw00f -l

Tip نکات مهم CTF

bash 
# 1. تشخیص با نمایش همه WAFهای احتمالی
wafw00f example.com | grep -i "detected"

# 2. تشخیص با فایل گزارش
wafw00f -r waf_report.html example.com

# 3. تشخیص با تست‌های خاص
wafw00f -T 3 example.com

# 4. تشخیص با proxy
wafw00f -p http://127.0.0.1:8080 example.com

# 5. تشخیص با User-Agent
wafw00f -H "User-Agent: Mozilla/5.0" example.com

# 6. تشخیص با فایل ورودی
echo "example.com" > targets.txt
wafw00f -f targets.txt

# 7. تشخیص با نمایش خلاصه
wafw00f example.com | tail -5

# 8. تشخیص با نمایش فقط نتیجه
wafw00f -q example.com

3️⃣ NIKTO - Web Server Scanner

📌 هدف: اسکن وب‌سرور برای پیدا کردن misconfiguration و فایل‌های حساس

Base دستورات پایه و ترکیبی

bash 
# ==================== دستورات اصلی ====================

# اسکن ساده
nikto -h example.com

# اسکن با SSL
nikto -h https://example.com -ssl

# اسکن با پورت خاص
nikto -h example.com -p 8080

# اسکن با خروجی HTML
nikto -h example.com -o report.html

# اسکن با خروجی XML
nikto -h example.com -o report.xml

# ==================== دستورات ترکیبی ====================

# اسکن با User-Agent خاص
nikto -h example.com -useragent "Mozilla/5.0"

# اسکن با Proxy
nikto -h example.com -useproxy http://127.0.0.1:8080

# اسکن با Timeout
nikto -h example.com -timeout 10

# اسکن با تنظیمات
nikto -h example.com -Tuning 12345

bash

4️⃣ GOBUSTER - Directory/File Enumeration ⭐⭐⭐⭐⭐

📌 هدف: پیدا کردن دایرکتوری‌ها و فایل‌های مخفی

Base دستورات پایه و ترکیبی

bash 
# ==================== دستورات اصلی ====================

# Directory Enumeration
gobuster dir -u http://example.com -w /usr/share/wordlists/dirb/common.txt

# File Enumeration با پسوند
gobuster dir -u http://example.com -w /usr/share/wordlists/dirb/common.txt -x php,txt,html

# با تعداد Thread بالا
gobuster dir -u http://example.com -w wordlist.txt -t 50

# با نمایش Status Code
gobuster dir -u http://example.com -w wordlist.txt -s 200,204,301,302,307,403

# با خروجی JSON
gobuster dir -u http://example.com -w wordlist.txt -o gobuster.json -j

# ==================== دستورات ترکیبی ====================

# با خروجی TXT
gobuster dir -u http://example.com -w wordlist.txt -o gobuster.txt

# با خروجی CSV
gobuster dir -u http://example.com -w wordlist.txt -o gobuster.csv

# با خروجی JSON
gobuster dir -u http://example.com -w wordlist.txt -o gobuster.json -j

# با نمایش فقط دایرکتوری‌ها
gobuster dir -u http://example.com -w wordlist.txt | grep -i "directory"

CTF سناریوهای CTF

bash 
# 1. اسکن دایرکتوری‌های رایج

gobuster dir -u http://example.com -w /usr/share/wordlists/dirb/common.txt -t 50 -s 200,204,301,302,307,403

gobuster dir -u http://example.com -w wordlist.txt -c "PHPSESSID=value"

# 7. اسکن با User-Agent خاص
gobuster dir -u http://example.com -w wordlist.txt -a "Mozilla/5.0"

# 8. اسکن با Proxy
gobuster dir -u http://example.com -w wordlist.txt -p http://127.0.0.1:8080

gobuster dir -u http://example.com -w wordlist.txt -t 50 -q

gobuster dir -u http://example.com -w wordlist.txt -t 50 -l

gobuster dir -u http://example.com -w /usr/share/wordlists/rockyou.txt -x php,txt

Tip نکات مهم CTF

5️⃣ FFUF - Web Fuzzing ⭐⭐⭐⭐⭐

📌 هدف: Fuzzing پیشرفته برای پیدا کردن دایرکتوری، فایل، پارامتر و VHOST

Base دستورات پایه و ترکیبی

bash 
# ==================== دستورات اصلی ====================

# Directory Fuzzing
ffuf -u http://example.com/FUZZ -w /usr/share/wordlists/dirb/common.txt

# File Fuzzing با پسوند
ffuf -u http://example.com/FUZZ -w /usr/share/wordlists/dirb/common.txt -e .php,.txt,.html

# با فیلتر Status Code
ffuf -u http://example.com/FUZZ -w wordlist.txt -fc 404,403

# با نمایش Status Code
ffuf -u http://example.com/FUZZ -w wordlist.txt -mc 200,204,301,302

# با خروجی JSON
ffuf -u http://example.com/FUZZ -w wordlist.txt -o ffuf.json -json

# ==================== دستورات ترکیبی ====================

# با تعداد Thread بالا
ffuf -u http://example.com/FUZZ -w wordlist.txt -t 50

# با خروجی CSV
ffuf -u http://example.com/FUZZ -w wordlist.txt -o ffuf.csv -of csv

# با Delay
ffuf -u http://example.com/FUZZ -w wordlist.txt -p 0.1

# با Timeout
ffuf -u http://example.com/FUZZ -w wordlist.txt -timeout 5

CTF سناریوهای CTF

# 16. با خروجی JSON

# 17. با خروجی CSV

# 18. با خروجی HTML

# 19. با نمایش کامل
ffuf -u http://example.com/FUZZ -w wordlist.txt -v

# 20. با verbose
ffuf -u http://example.com/FUZZ -w wordlist.txt -v -c

6️⃣ DIRB - Directory Enumeration

📌 هدف: پیدا کردن دایرکتوری‌ها با Brute Force
bash

dirb http://example.com

# با پسوند
dirb http://example.com -X .php,.txt,.html

# با خروجی
dirb http://example.com -o dirb.txt

# ==================== دستورات ترکیبی ====================

# با User-Agent

#!/bin/bash
# web_cheatsheet.sh - اسکریپت کامل Web Recon

TARGET=${1:-example.com}
OUTPUT_DIR="web_recon_$(date +%Y%m%d_%H%M%S)"
mkdir -p $OUTPUT_DIR

echo "========================================="
echo "  WEB RECON CHEAT SHEET AUTO-RUN"
echo "========================================="

# 1. WhatWeb
echo "[1/8] Running WhatWeb..."
whatweb -a 3 $TARGET > $OUTPUT_DIR/01_whatweb.txt
echo "✓ WhatWeb saved"

# 2. Wafw00f
echo "[2/8] Running Wafw00f..."
wafw00f $TARGET > $OUTPUT_DIR/02_wafw00f.txt
echo "✓ Wafw00f saved"

# 3. Nikto
echo "[3/8] Running Nikto..."
nikto -h $TARGET -ssl > $OUTPUT_DIR/03_nikto.txt
echo "✓ Nikto saved"

# 4. Gobuster
echo "[4/8] Running Gobuster..."
gobuster dir -u http://$TARGET -w /usr/share/wordlists/dirb/common.txt -t 50 > $OUTPUT_DIR/04_gobuster.txt
gobuster dir -u http://$TARGET -w /usr/share/wordlists/dirb/common.txt -x php,txt,html -t 50 > $OUTPUT_DIR/04_gobuster_files.txt
echo "✓ Gobuster saved"

# 5. FFUF
echo "[5/8] Running FFUF..."
ffuf -u http://$TARGET/FUZZ -w /usr/share/wordlists/dirb/common.txt -fc 404,403 -t 50 > $OUTPUT_DIR/05_ffuf.txt
ffuf -u http://$TARGET/FUZZ -w /usr/share/wordlists/dirb/common.txt -e .php,.txt,.html -fc 404,403 -t 50 > $OUTPUT_DIR/05_ffuf_files.txt
echo "✓ FFUF saved"

# 6. DIRB
echo "[6/8] Running DIRB..."
dirb http://$TARGET -o $OUTPUT_DIR/06_dirb.txt
echo "✓ DIRB saved"

# 7. Dirsearch
echo "[7/8] Running Dirsearch..."
dirsearch -u http://$TARGET -e php,txt,html -o $OUTPUT_DIR/07_dirsearch.txt
echo "✓ Dirsearch saved"

# 8. CURL
echo "[8/8] Running CURL..."
curl -I -L $TARGET > $OUTPUT_DIR/08_headers.txt
curl $TARGET/robots.txt > $OUTPUT_DIR/08_robots.txt 2>/dev/null
curl $TARGET/sitemap.xml > $OUTPUT_DIR/08_sitemap.xml 2>/dev/null
echo "✓ CURL saved"

echo ""
echo "========================================="
echo "  ✅ RECON COMPLETE"
echo "========================================="
echo "📁 Output: $OUTPUT_DIR"
ls -la $OUTPUT_DIR
